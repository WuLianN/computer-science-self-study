示例一 · 最大公约数

采用的题意：按原题前文的“最大公约数”实现。原文输出要求另写“最小公约数”，这一差异仍保留在大纲疑点说明中。

本例输入约定：两个在 C 的 long 范围内的整数，允许负数，但不能同时为 0；输出两数绝对值的最大公约数。超出 long 范围的数字不属于本例输入约定。

  1. 用 magnitude 取得非负幅值，再调用最大公约数函数。
  2. 当 b != 0，计算余数 a % b，把原来的 b 和余数作为下一组数。
  3. b 变成 0 时返回 a；两个数同时为 0 或读取失败时输出错误提示。

下载 gcd.c · 单独编译本文件。

#include <stdio.h>

unsigned long magnitude(long value)
{
    if (value < 0) {
        return 0UL - (unsigned long)value;
    }
    return (unsigned long)value;
}

unsigned long gcd(unsigned long a, unsigned long b)
{
    unsigned long remainder;

    while (b != 0) {
        remainder = a % b;
        a = b;
        b = remainder;
    }
    return a;
}

int main(void)
{
    long a;
    long b;
    unsigned long result;

    if (scanf("%ld %ld", &a, &b) != 2) {
        fprintf(stderr, "Expected two integers.\n");
        return 1;
    }
    if (a == 0 && b == 0) {
        fprintf(stderr, "At least one integer must be nonzero.\n");
        return 1;
    }

    result = gcd(magnitude(a), magnitude(b));
    printf("%lu\n", result);
    return 0;
}

示例二 · 学生成绩处理

对应原文：用结构体保存学号、姓名和成绩，分别编写输入、输出、统计总分和最高分的函数，并在主函数测试。

本站补充约定：原文未给出输入格式和科目数。本例首行输入学生数 n 与每人科目数 m，随后每行输入学号、姓名和 m 个成绩，逐个学生计算总分与最高科目分。m=1 时也可作为单科练习。

本例限制：1≤n≤100、1≤m≤10，成绩为 0–100 的有限数值；学号不超过 19 字节、姓名不超过 31 字节且不含空格。示例采用英文姓名便于核对输出；实际格式按题面调整。

  1. Student 结构体保存学号、姓名和成绩数组，学号使用字符串以保留前导 0。
  2. input_students 读取并检查输入；学生数、科目数和成绩均作范围检查。
  3. total_score 从 0 开始累加；highest_score 从第一科成绩开始比较。
  4. output_students 调用两个统计函数输出每位学生的结果，主函数负责串起整个流程。

下载 student_scores.c · 此文件有自己的 main，和 gcd.c 分开编译。

#include <stdio.h>

#define MAX_STUDENTS 100
#define MAX_SUBJECTS 10

typedef struct {
    char id[20];
    char name[32];
    double score[MAX_SUBJECTS];
} Student;

int input_students(Student students[], int count, int subjects)
{
    int i;
    int j;

    for (i = 0; i < count; ++i) {
        if (scanf("%19s %31s", students[i].id, students[i].name) != 2) {
            return 0;
        }
        for (j = 0; j < subjects; ++j) {
            if (scanf("%lf", &students[i].score[j]) != 1) {
                return 0;
            }
            if (!(students[i].score[j] >= 0.0 &&
                  students[i].score[j] <= 100.0)) {
                return 0;
            }
        }
    }
    return 1;
}

double total_score(const Student *student, int subjects)
{
    double total;
    int j;

    total = 0.0;
    for (j = 0; j < subjects; ++j) {
        total += student->score[j];
    }
    return total;
}

double highest_score(const Student *student, int subjects)
{
    double highest;
    int j;

    highest = student->score[0];
    for (j = 1; j < subjects; ++j) {
        if (student->score[j] > highest) {
            highest = student->score[j];
        }
    }
    return highest;
}

void output_students(const Student students[], int count, int subjects)
{
    int i;
    int j;

    printf("ID NAME SCORES TOTAL MAX\n");
    for (i = 0; i < count; ++i) {
        printf("%s %s", students[i].id, students[i].name);
        for (j = 0; j < subjects; ++j) {
            printf(" %.2f", students[i].score[j]);
        }
        printf(" total=%.2f max=%.2f\n",
               total_score(&students[i], subjects),
               highest_score(&students[i], subjects));
    }
}

int main(void)
{
    Student students[MAX_STUDENTS];
    int count;
    int subjects;

    if (scanf("%d %d", &count, &subjects) != 2 ||
        count < 1 || count > MAX_STUDENTS ||
        subjects < 1 || subjects > MAX_SUBJECTS) {
        fprintf(stderr, "Invalid student or subject count.\n");
        return 1;
    }
    if (!input_students(students, count, subjects)) {
        fprintf(stderr, "Invalid student data.\n");
        return 1;
    }

    output_students(students, count, subjects);
    return 0;
}

易错点：数组下标范围为 0 到 m−1；scanf 读取 double 用 %lf；指向结构体的指针通过 -> 访问成员。本例统计的是每个学生，不是把全班所有人的成绩混成一个总分。
